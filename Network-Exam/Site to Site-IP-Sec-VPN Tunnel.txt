Router 1
int fa0/0
ip add 192.168.1.1 255.255.255.0
no sh

! Phase-1 (ISAKMP / IKE v1)
crypto isakmp policy 10
 hash sha
 authentication pre-share
 group 2
 lifetime 86400
 exit
crypto isakmp key MyRouting address 2.2.2.2

! Phase-2 (IPSec Transform Set)
crypto ipsec transform-set MyRouting esp-sha-hmac esp-3des
mode tunnel
exit


! Access‐list untuk traffic yang dienkrip
ip access-list ex TUNNEL-TRAFIC
permit ip 192.168.1.0 0.0.0.255 192.168.2.0 0.0.0.255
exit

! Crypto map
crypto map MyRouting-VPN 10 ipsec-isakmp
 set peer 2.2.2.2
 set transform-set MyRouting
 match address TUNNEL-TRAFFIC
exit

! Terapkan crypto map ke interface publik
interface fa0/1       ! sesuaikan nama interface publik ke Internet
 ip address 1.1.1.1 255.255.255.252  ! contoh
 no sh
 crypto map MyRouting-VPN

! Routing lokal/rule firewall jika perlu
ip route 192.168.2.0 255.255.255.0 2.2.2.2



Verifikasi
show crypto session
