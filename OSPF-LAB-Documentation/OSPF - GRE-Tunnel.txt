Konfigurasi  Router1

Router1(config-if)#desc TO Router2
Router1(config-if)#ip add 12.12.12.1 255.255.255.0
Router1(config-if)#no sh
Router1(config-if)#int lo0
Router1(config-if)#ip add 1.1.1.1 255.255.255.255
Router1(config-if)#router ospf 1
Router1(config-router)#net 1.1.1.1 0.0.0.0 are 0
Router1(config-router)#net 12.12.12.0 0.0.0.255 are 1
