CCNP - BGP Remove Private AS


Konfigurasi Router1
hostname Router1
!
interface fa0/0 
  ip address 12.12.12.1 255.255.255.0
  no shutdown
!
router bgp 1
  neighbor 12.12.12.2 remote-as 64512
  network 12.12.12.0 mask 255.255.255.0

Konfigurasi Router2
hostname Router2
!
interface fa0/0
  ip address 12.12.12.2 255.255.255.0
  no shutdown
interface fa0/1
  ip address 23.23.23.2 255.255.255.0
  no shutdown
!
router bgp 64512
neighbor 12.12.12.1 remote-as 1
neighbor 23.23.23.3 remote-as 65000

 
Konfigurasi Router3

hostname Router3
!
interface fa0/0
  ip address 23.23.23.3 255.255.255.0
  no shutdown
!
int lo0
ip add 3.3.3.3 255.255.255.255

router bgp 65000
  neighbor 23.23.23.2 remote-as 64512
  network 23.23.23.0 mask 255.255.255.0
