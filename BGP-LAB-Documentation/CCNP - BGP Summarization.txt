int lo0

access-list 1 permit host 1.1.1.1   
route-map NO-EXPORT permit 10
match ip address 1
set community no-export

router bgp 1
bgp confederation identifier 123
bgp confederation peers 23
neighbor 12.12.12.2 remote-as 23
net 1.1.1.1 mask 255.255.255.255


access-list 2 permit 1.1.1.1
route-map LOCAL-AS permit 10
match ip address 2
set community local-as
route-map LOCAL-AS permit 20

router bgp 1
neighbor 12.12.12.2 route-map LOCAL-AS out
neighbor 12.12.12.2 send-community
